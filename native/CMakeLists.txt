option(DXSHARP_HYDRA_PROBE "Compile Houdini/OpenUSD Hydra ABI probe" OFF)
if(DXSHARP_HYDRA_PROBE)
  add_executable(hydra_abi_probe hydra/abi_probe.cpp)
  target_compile_features(hydra_abi_probe PRIVATE cxx_std_17)
  # Match the CPython ABI shipped with the selected Houdini SDK.
  # Do not allow FindPython to select an unrelated system installation.
  get_filename_component(_hfs "${Houdini_DIR}/../.." ABSOLUTE)
  file(GLOB_RECURSE _python_candidates LIST_DIRECTORIES false "${_hfs}/*python*.lib")
  set(_python_imports "")
  foreach(_lib IN LISTS _python_candidates)
    get_filename_component(_name "${_lib}" NAME)
    if(_name MATCHES "^python3[0-9]+[.]lib$")
      get_filename_component(_stem "${_lib}" NAME_WE)
      if(EXISTS "${_hfs}/bin/${_stem}.dll")
        list(APPEND _python_imports "${_lib}")
      else()
        file(GLOB_RECURSE _runtime LIST_DIRECTORIES false "${_hfs}/*${_stem}.dll")
        if(_runtime)
          list(APPEND _python_imports "${_lib}")
        endif()
      endif()
    endif()
  endforeach()
  list(REMOVE_DUPLICATES _python_imports)
  set(DXSHARP_HOUDINI_PYTHON_LIBRARY "" CACHE FILEPATH "Exact Houdini CPython import library override")
  if(NOT DXSHARP_HOUDINI_PYTHON_LIBRARY)
    list(LENGTH _python_imports _count)
    if(NOT _count EQUAL 1)
      message(FATAL_ERROR "Expected one bundled CPython import library; candidates: ${_python_imports}. Set DXSHARP_HOUDINI_PYTHON_LIBRARY explicitly.")
    endif()
    list(GET _python_imports 0 DXSHARP_HOUDINI_PYTHON_LIBRARY)
  endif()
  if(NOT EXISTS "${DXSHARP_HOUDINI_PYTHON_LIBRARY}")
    message(FATAL_ERROR "CPython import library not found: ${DXSHARP_HOUDINI_PYTHON_LIBRARY}")
  endif()
  message(STATUS "Hydra CPython import library: ${DXSHARP_HOUDINI_PYTHON_LIBRARY}")
  target_link_libraries(hydra_abi_probe PRIVATE Houdini "${DXSHARP_HOUDINI_PYTHON_LIBRARY}")
endif()
